-- Devuelve la ruta del archivo de sonido (primero la ruta fija), o nil si no existe
function get_file_path(filename)
 local p=carpetaSonidos.."/"..filename
 if file_exists(p) then return p end
 local d=nil
 if type(get_script_dir)=="function" then pcall(function() d=get_script_dir() end) end
 if type(d)=="string" and d~="" then
  if d:sub(-1)~="/" then d=d.."/" end
  if file_exists(d..filename) then return d..filename end
 end
 return nil
end

  mp.setOnCompletionListener(MediaPlayer.OnCompletionListener{
   onCompletion=function()
    -- Respaldo: si por algún motivo no hizo bucle, vuelve a empezar

     pcall(function() mp.seekTo(0) mp.start() end)

   end
  })
  mp.start()

 end)

end

function nuevoBuffer(n)
 local ok,b=pcall(function() return byte[n] end)
 if ok and b then return b end
 return Array.newInstance(Byte.TYPE,n)
end

-- Descarga el .zip y lo extrae SIEMPRE en la carpeta "Sonidos del juego"
function descargarEfectos()
 vibrateDevice(200)
 if descargandoEfectos then
  hablar("Ya se están descargando los efectos de sonido")
  return
 end
 descargandoEfectos=true
 hablar("Descargando efectos de sonido. Espera por favor")
 local tmp=basePath.."/sonidos_temp.zip"
 local destino=carpetaSonidos

 Thread(Runnable{run=function()
  local ok,err=pcall(function()
   File(basePath).mkdirs()
   File(destino).mkdirs()

   -- 1) Descarga binaria del zip
   local con=URL(SONIDOS_URL).openConnection()
   con.setConnectTimeout(20000)
   con.setReadTimeout(60000)
   con.setInstanceFollowRedirects(true)
   con.connect()
   local code=con.getResponseCode()
   if code~=200 then error("http "..tostring(code)) end
   local input=con.getInputStream()
   local out=FileOutputStream(tmp)
   local buf=nuevoBuffer(8192)
   local n=input.read(buf)
   while n~=-1 do
    out.write(buf,0,n)
    n=input.read(buf)
   end
   out.close()
   input.close()
   con.disconnect()

   -- 2) Validar que sea un zip (empieza con "PK")
   local fis=FileInputStream(tmp)
   local b1=fis.read()
   local b2=fis.read()
   fis.close()
   if b1~=80 or b2~=75 then error("no es un zip") end

   -- 3) Extraer
   local canonDestino=File(destino).getCanonicalPath()
   local zis=ZipInputStream(FileInputStream(tmp))
   local entry=zis.getNextEntry()
   while entry~=nil do
    local nombre=tostring(entry.getName())
    nombre=(nombre:gsub("\\","/"))
    nombre=(nombre:gsub("^Sonidos del juego/",""))
    if nombre~="" and not entry.isDirectory() then
     local f=File(destino.."/"..nombre)
     local canon=f.getCanonicalPath()
     if canon:sub(1,#canonDestino)==canonDestino then
      f.getParentFile().mkdirs()
      local fo=FileOutputStream(f)
      local m=zis.read(buf)
      while m~=-1 do
       fo.write(buf,0,m)
       m=zis.read(buf)
      end
      fo.close()
     end
    end
    zis.closeEntry()
    entry=zis.getNextEntry()
   end
   zis.close()
   File(tmp).delete()

   -- 4) Si el zip traía otra carpeta contenedora, subir su contenido
   if not File(destino.."/sonido.lua").exists() then
    local hijos=File(destino).listFiles()
    if hijos then
     for i=0,#hijos-1 do
      local h=hijos[i]
      if h.isDirectory() and File(h.getPath().."/sonido.lua").exists() then
       local contenido=h.listFiles()
       for j=0,#contenido-1 do
        contenido[j].renameTo(File(destino.."/"..contenido[j].getName()))
       end
      end
     end
    end
   end
   if not File(destino.."/sonido.lua").exists() then error("falta sonido.lua") end
  end)

  handler.post(Runnable{run=function()
   descargandoEfectos=false
   if ok then
    pcall(cargarModuloSonido)
    hablar("Efectos de sonido descargados e instalados")
   else
    pcall(function() File(tmp).delete() end)
    hablar("Error al descargar los efectos de sonido")
   end
  end})
 end}).start()
end

-- Acepta dd/mm/aaaa (formato del .txt) y también mm/dd/aaaa cuando el mes es evidente
function fechaATimestamp(f)
 local a,b,y=f:match("(%d+)/(%d+)/(%d+)")
 if not a then return 0 end
 a,b,y=tonumber(a),tonumber(b),tonumber(y)
 local d,m=a,b
 if b>12 and a<=12 then d,m=b,a end
 return os.time{year=y,month=m,day=d}
end

function parsearActualizacion(c)
 local lineas,fecha,cambios,enlace={},nil,{},nil
 for l in c:gmatch("[^\r\n]+") do table.insert(lineas,l) end
 if #lineas<2 then return nil end
 fecha=lineas[1]:match("^%s*(.-)%s*$")
 for i=2,#lineas do
  local l=lineas[i]:match("^%s*(.-)%s*$")
  if l:match("^https?://") then
   enlace=l
  elseif l~="" and not enlace then
   local cb=l:gsub("^%-+%s*","")
   if cb~="" and #cb>3 then table.insert(cambios,cb) end
  end
 end
 return {fecha=fecha,cambios=cambios,enlace=enlace}
end

function guardarArchivo(ruta,contenido)
 return pcall(function()
  local dir=File(ruta).getParent()
  if dir and not File(dir).exists() then File(dir).mkdirs() end
  local f=io.open(ruta,"w")
  f:write(contenido)
  f:close()
 end)
end

function verificarActualizacion()
 local resuelto=false
 local function continuar(info)
  if resuelto then return end
  resuelto=true
  local hay=false
  pcall(function()
   if info and info.fecha and fechaATimestamp(info.fecha)>fechaATimestamp(FECHA_CODIGO) then hay=true end
  end)
  if hay then
   mostrarDialogo(info)   -- hay versión nueva: NO se muestra el juego
  else
   if nombreUsuario==nil or tostring(nombreUsuario)=="" or paisUsuario==nil or tostring(paisUsuario)=="" then
    mostrarRegistro()
   else
    iniciarJuego()
   end
  end
 end
 local ok=pcall(function()
  Http.get(UPDATES_URL,function(s,d)
   local info=nil
   if s==200 and d and d~="" then
    pcall(function() info=parsearActualizacion(d) end)
   end
   handler.post(Runnable{run=function() continuar(info) end})
  end)
 end)
 if not ok then
  continuar(nil)
  return
 end
 -- Sin respuesta en 8 segundos: se abre el juego normalmente
 handler.postDelayed(Runnable{run=function() continuar(nil) end},8000)
end

function vibrateDevice(duration)
 if not vibracionActiva then return end
 local vibrator=ctx.getSystemService(Context.VIBRATOR_SERVICE)
 if vibrator and vibrator.hasVibrator() then
  local escala=({["Suave"]=0.35,["Normal"]=0.65,["Fuerte"]=0.85,["Muy fuerte"]=1.0})[intensidadVibracion] or 0.65
  local dur=math.max(20,math.floor(duration*escala))
  if Build.VERSION.SDK_INT>=26 then
   vibrator.vibrate(VibrationEffect.createOneShot(dur, math.max(1,math.floor(255*escala))))
  else vibrator.vibrate(dur) end
 end
end

function updateTTSSpeed()
 if tts then
  local finalSpeed = selectedSpeedIndex * (selectedSpeed / 100)
  if finalSpeed < 0.1 then finalSpeed = 0.1 end
  if finalSpeed > 10.0 then finalSpeed = 10.0 end
  tts.setSpeechRate(finalSpeed)
 end
end

function iniciarTTS()
 if tts ~= nil then
  pcall(function() tts.stop() end)
  pcall(function() tts.shutdown() end)
  tts = nil
 end
 tts=TextToSpeech(ctx,TextToSpeech.OnInitListener{
  onInit=function(s)
   if s==TextToSpeech.SUCCESS then
    tts.setLanguage(Locale("es","ES"))
    tts.setPitch(1.0)
    if ttsMotor~="" and ttsMotor~=nil then
     pcall(function() tts.setEngineByPackageName(ttsMotor) end)
    end
    updateTTSSpeed()
    handler.postDelayed(function()
     mostrarMenu()
    end,400)
   end
  end
 }, ttsMotor)
end

function hablar(texto)
 if tts~=nil then
  tts.speak(texto,TextToSpeech.QUEUE_FLUSH,nil)
 end
end

function cartaUnicaUsada(nombre)
 for _,c in ipairs(cartasUnicasUsadas) do
  if c==nombre then return true end
 end
 return false
end

function marcarCartaUnica(nombre)
 if not cartaUnicaUsada(nombre) then
  table.insert(cartasUnicasUsadas, nombre)
 end
end

function esCartaUnica(nombre)
 for _,c in ipairs(CARTAS_UNICAS) do
  if c==nombre then return true end
 end
 return false
end

function robarCarta()
 local r=math.random(100)
 if not cartaUnicaUsada("Plan_Dental") and r<=5 then return "Plan_Dental" end
 if not cartaUnicaUsada("Guantes_Antiadherentes") and r<=10 then return "Guantes_Antiadherentes" end
 if not cartaUnicaUsada("Bolsas_Antirrobo") and r<=15 then return "Bolsas_Antirrobo" end
 if not cartaUnicaUsada("Control_Azucar") and r<=20 then return "Control_Azucar" end

 if r<=60 then
  local valores={25,50,75,100,150,175,200}
  return valores[math.random(#valores)]
 elseif r<=70 then
  return "Ladron_Caramelos"
 elseif r<=80 then
  return "Manos_Pegajosas"
 elseif r<=85 then
  return "Visita_Dentista"
 elseif r<=92 then
  return "Bajon_Azucar"
 elseif r<=99 then
  return "Subidon_Azucar"
 else
  return "Escudo"
 end
end

function rellenarMano(mano)
 while #mano < getTamanoMano() do
  local nueva = robarCarta()
  local intentos=0
  while esCartaUnica(nueva) and cartaUnicaUsada(nueva) and intentos<50 do
   nueva = robarCarta()
   intentos=intentos+1
  end
  if esCartaUnica(nueva) and cartaUnicaUsada(nueva) then
   nueva = 25
  end
  table.insert(mano, nueva)
 end
end

function asegurarDesbloqueo(mano, tipoBloqueo)
 local tiene=false
 for _,carta in ipairs(mano) do
  if type(carta)=="string" and carta=="Desbloqueo_"..tipoBloqueo then
   tiene=true
   break
  end
 end
 if not tiene then
  mano[math.random(#mano)] = "Desbloqueo_"..tipoBloqueo
 end
end

function ejecutarEventos(quien)
 eventos.contadorTurnos = eventos.contadorTurnos + 1
 local tienePlan = tieneDefensa(quien=="jugador" and defensasJugador or defensasBot, "Plan_Dental", false)

 if not eventos.dentistaUsado and eventos.contadorTurnos >= 2 then
  eventos.dentistaUsado = true
  if not tienePlan then
   local perdida = math.random(200,400)
   if quien=="jugador" then
    caramelosJugador = caramelosJugador - perdida
    if caramelosJugador < 0 then caramelosJugador = 0 end
    hablar("Evento: Visita al dentista. Pierdes "..perdida.." caramelos")
   else
    caramelosBot = caramelosBot - perdida
    if caramelosBot < 0 then caramelosBot = 0 end
    hablar("Evento: Bot visita al dentista. Pierde "..perdida.." caramelos")
   end
  else
   hablar("Evento: Visita al dentista. Tu Plan Dental te protege")
  end
 end

 if eventos.feriaUsadas < 2 and eventos.contadorTurnos % math.random(2,3) == 0 then
  eventos.feriaUsadas = eventos.feriaUsadas + 1
  local ganancia = math.random(350,600)
  if quien=="jugador" then
   caramelosJugador = caramelosJugador + ganancia
   caramelosTotalesRecolectados = caramelosTotalesRecolectados + ganancia
   hablar("Evento: Feria de caramelos. Ganas "..ganancia.." caramelos")
  else
   caramelosBot = caramelosBot + ganancia
   hablar("Evento: Feria de caramelos. Bot gana "..ganancia.." caramelos")
  end
 end

 if not eventos.lluviaUsada and eventos.contadorTurnos >= 5 then
  eventos.lluviaUsada = true
  local perdida = math.random(50,100)
  if quien=="jugador" then
   caramelosJugador = caramelosJugador - perdida
   if caramelosJugador < 0 then caramelosJugador = 0 end
   hablar("Evento: Lluvia de caramelos. Pierdes "..perdida.." caramelos")
  else
   caramelosBot = caramelosBot - perdida
   if caramelosBot < 0 then caramelosBot = 0 end
   hablar("Evento: Lluvia de caramelos. Bot pierde "..perdida.." caramelos")
  end
 end
end

function aplicarBloqueo(objetivo, tipo, quien)
 if objetivo=="jugador" then
  if tipo=="Pegajoso" and tieneDefensa(defensasJugador,"Guantes_Antiadherentes",false) then
   hablar("Guantes antiadherentes te protegen de manos pegajosas")
   return
  end
  if tipo=="Ladron" and tieneDefensa(defensasJugador,"Bolsas_Antirrobo",false) then
   hablar("Bolsas antirrobo te protegen del ladrón")
   return
  end
  if tipo=="Dentista" and tieneDefensa(defensasJugador,"Plan_Dental",false) then
   hablar("Plan dental te protege del dentista")
   return
  end
  if tipo=="Azucar" and tieneDefensa(defensasJugador,"Control_Azucar",false) then
   hablar("Control de azúcar te protege del bajón")
   return
  end

  bloqueadoJugador=true
  tipoBloqueoJugador=tipo
  asegurarDesbloqueo(manoJugador, tipo)
  hablar(quien.." te bloquea con "..tipo)
 else
  if tipo=="Pegajoso" and tieneDefensa(defensasBot,"Guantes_Antiadherentes",false) then
   hablar("Bot tiene guantes antiadherentes")
   return
  end
  if tipo=="Ladron" and tieneDefensa(defensasBot,"Bolsas_Antirrobo",false) then
   hablar("Bot tiene bolsas antirrobo")
   return
  end
  if tipo=="Dentista" and tieneDefensa(defensasBot,"Plan_Dental",false) then
   hablar("Bot tiene plan dental")
   return
  end
  if tipo=="Azucar" and tieneDefensa(defensasBot,"Control_Azucar",false) then
   hablar("Bot tiene control de azúcar")
   return
  end

  bloqueadoBot=true
  tipoBloqueoBot=tipo
  asegurarDesbloqueo(manoBot, tipo)
  hablar("Tu bloqueas al bot con "..tipo)
 end
end

function tieneDefensa(defensas, tipo, consumir)
 consumir = consumir ~= false
 for i,def in ipairs(defensas) do
  if def==tipo then
   if consumir then
    table.remove(defensas,i)
   end
   return true
  end
 end
 return false
end

function jugarCartaSeleccionada(indice)
 if not enArena or not turnoJugador then return end
 vibrar({100})

 local carta=manoJugador[indice]
 table.remove(manoJugador,indice)
 sonidoCarta(carta)

 if type(carta)=="string" and esCartaUnica(carta) then
  marcarCartaUnica(carta)
 end

 rellenarMano(manoJugador)

 local mensajeJugador=""

 if type(carta)=="number" then
  local cantidad = carta
  if subidonActivoJugador and cantidad > 50 then
   cantidad = 50
   mensajeJugador = "Exceso de dulce. Solo sumas 50 caramelos. Total "..caramelosJugador+50
  else
   mensajeJugador = "Sumas "..cantidad.." caramelos. Total "..caramelosJugador+cantidad
  end
  caramelosJugador=caramelosJugador+cantidad
  caramelosTotalesRecolectados=caramelosTotalesRecolectados+cantidad
  subidonActivoJugador=false

 elseif type(carta)=="string" then
  if carta=="Ladron_Caramelos" then
   if tieneDefensa(defensasBot,"Bolsas_Antirrobo",false) then
    mensajeJugador = "El bot tiene bolsas antirrobo. No puedes robarle"
   else
    local robo=math.random(50,150)
    if caramelosBot>=robo then
     caramelosBot=caramelosBot-robo
     caramelosJugador=caramelosJugador+robo
     caramelosTotalesRecolectados=caramelosTotalesRecolectados+robo
     mensajeJugador = "Le robas "..robo.." caramelos al bot"
    else
     caramelosJugador=caramelosJugador+caramelosBot
     caramelosTotalesRecolectados=caramelosTotalesRecolectados+caramelosBot
     mensajeJugador = "Le robas todos los caramelos al bot"
     caramelosBot=0
    end
   end

  elseif carta=="Manos_Pegajosas" then
   aplicarBloqueo("bot","Pegajoso","Tu")
   mensajeJugador = "Usas Manos Pegajosas"

  elseif carta=="Visita_Dentista" then
   if caramelosBot>0 then
    local perdida=math.random(200,400)
    caramelosBot=caramelosBot-perdida
    if caramelosBot<0 then caramelosBot=0 end
    mensajeJugador = "Bot visita al dentista. Pierde "..perdida.." caramelos"
   else
    mensajeJugador = "Bot no tiene caramelos para perder"
   end

  elseif carta=="Control_Azucar" then
   table.insert(defensasJugador,"Control_Azucar")
   mensajeJugador = "Control de azúcar activado"

  elseif carta=="Bajon_Azucar" then
   aplicarBloqueo("bot","Azucar","Tu")
   mensajeJugador = "Usas Bajón de Azúcar"

  elseif carta=="Subidon_Azucar" then
   subidonActivoJugador=true
   aplicarBloqueo("bot","Azucar","Tu")
   mensajeJugador = "Subidón de Azúcar. Tus caramelos +50 quedan limitados a 50"

  elseif carta=="Guantes_Antiadherentes" then
   table.insert(defensasJugador,"Guantes_Antiadherentes")
   mensajeJugador = "Guantes antiadherentes equipados"

  elseif carta=="Bolsas_Antirrobo" then
   table.insert(defensasJugador,"Bolsas_Antirrobo")
   mensajeJugador = "Bolsas antirrobo equipadas"

  elseif carta=="Plan_Dental" then
   table.insert(defensasJugador,"Plan_Dental")
   mensajeJugador = "Plan dental activado"

  elseif carta:find("Defensa_") then
   local tipo=carta:gsub("Defensa_","")
   table.insert(defensasJugador,tipo)
   mensajeJugador = "Te proteges contra "..tipo

  elseif carta:find("Desbloqueo_") then
   local tipo=carta:gsub("Desbloqueo_","")
   if tipoBloqueoJugador==tipo then
    bloqueadoJugador=false
    tipoBloqueoJugador=""
    mensajeJugador = "Desbloqueado de "..tipo
   else
    mensajeJugador = "No sirve para "..tipoBloqueoJugador
   end

  elseif carta=="Escudo" then
   defensasJugador={"Escudo"}
   mensajeJugador = "Escudo activado"
  end
 end

 hablar(mensajeJugador)
 turnoJugador=false
 actualizarUI()
 mostrarMano()

 verificarVictoria()
 if not enArena then return end

 ejecutarEventos("jugador")
 actualizarUI()

 handler.removeCallbacksAndMessages(nil)
 handler.postDelayed(function()
  turnoBot()
 end,1200)
end

 local carta=manoBot[idx]

     caramelosJugador=caramelosJugador-robo

     mensajeBot = "Bot te roba todos tus caramelos"

  elseif carta=="Manos_Pegajosas" then

   if caramelosJugador>0 then

    caramelosJugador=caramelosJugador-perdida
    if caramelosJugador<0 then caramelosJugador=0 end
    mensajeBot = "Visitas al dentista. Pierdes "..perdida.." caramelos"
   else

   end

  elseif carta=="Control_Azucar" then
   table.insert(defensasBot,"Control_Azucar")
   mensajeBot = "Bot activa control de azúcar"

  elseif carta=="Bajon_Azucar" or carta=="Subidon_Azucar" then
   if carta=="Subidon_Azucar" then subidonActivoBot=true end
   aplicarBloqueo("jugador","Azucar","Bot")
   mensajeBot = "Bot usa "..carta

  elseif carta=="Guantes_Antiadherentes" then
   table.insert(defensasBot,"Guantes_Antiadherentes")
   mensajeBot = "Bot equipa guantes antiadherentes"

  elseif carta=="Bolsas_Antirrobo" then
   table.insert(defensasBot,"Bolsas_Antirrobo")
   mensajeBot = "Bot equipa bolsas antirrobo"

  elseif carta=="Plan_Dental" then
   table.insert(defensasBot,"Plan_Dental")
   mensajeBot = "Bot activa plan dental"

  elseif carta:find("Defensa_") then
   local tipo=carta:gsub("Defensa_","")
   table.insert(defensasBot,tipo)
   mensajeBot = "Bot se protege contra "..tipo

  elseif carta=="Escudo" then
   defensasBot={"Escudo"}
   mensajeBot = "Bot activa Escudo"
  end
 end

 hablar(mensajeBot)

function getAvailableEngines()
 local engines = {"Sistema"}
 local pm = ctx.getPackageManager()
 local intent = Intent(TextToSpeech.Engine.ACTION_CHECK_TTS_DATA)
 local list = pm.queryIntentActivities(intent, PackageManager.MATCH_DEFAULT_ONLY)
 for i = 0, list.size() - 1 do
  local pkg = list.get(i).activityInfo.packageName
  if pkg ~= "com.google.android.tts" then
   table.insert(engines, pkg)
  end
 end
 table.insert(engines, 2, "com.google.android.tts")
 return engines
end

function crearParamsVentana()
 local tipo
 if esAccesibilidad then
  tipo=2032 -- TYPE_ACCESSIBILITY_OVERLAY (no requiere permiso extra)
 elseif Build.VERSION.SDK_INT >= 26 then
  tipo=2038 -- TYPE_APPLICATION_OVERLAY
 else
  tipo=2002 -- TYPE_PHONE
 end
 local lp=WindowManager.LayoutParams()
 lp.width=-1
 lp.height=-1
 lp.type=tipo
 lp.flags=0
 lp.format=PixelFormat.TRANSLUCENT
 return lp
end

-- Muestra diálogos correctamente desde el servicio de accesibilidad.
-- Un Dialog normal intenta usar una ventana de aplicación y provoca BadTokenException.
function mostrarDialogoSeguro(dialogo)
 if dialogo == nil then return false end
 local ok = pcall(function()
  local ventana = dialogo.getWindow()
  if ventana ~= nil then
   local tipo
   if esAccesibilidad then
    tipo = 2032 -- TYPE_ACCESSIBILITY_OVERLAY
   elseif Build.VERSION.SDK_INT >= 26 then
    tipo = 2038 -- TYPE_APPLICATION_OVERLAY
   else
    tipo = 2002 -- TYPE_PHONE
   end
   ventana.setType(tipo)
  end
  dialogo.show()
 end)
 if not ok then
  -- Respaldo para servicios que no admiten ventanas Dialog: mostrar la vista como overlay.
  pcall(function()
   local vista = dialogo.getWindow().getDecorView()
   if vista ~= nil then
    wm.addView(vista, crearParamsVentana())
   end
  end)
 end
 return ok
end

 enMenu=false
 enArena=false

 cerrarConfig(true)
 cerrarPanelActualizacion()

 detenerEfecto()

 if not esAccesibilidad then
  pcall(function() ctx.stopSelf() end)
 end
end

-- ============================================================
-- Firebase Realtime Database: perfiles, comentarios y clasificación
-- Todas las peticiones de red se ejecutan en segundo plano.
-- ============================================================
function firebaseRequest(path, method, payload, callback)
 Thread(Runnable{run=function()
  local ok,result=pcall(function()
   local url=URL(FIREBASE_URL..path..".json")
   local con=url.openConnection()
   con.setConnectTimeout(10000); con.setReadTimeout(10000)
   con.setRequestMethod(method or "GET")
   if payload~=nil then
    con.setDoOutput(true)
    con.setRequestProperty("Content-Type","application/json; charset=utf-8")
    local out=con.getOutputStream()
    out.write(String(cjson.encode(payload)).getBytes("UTF-8")); out.flush(); out.close()
   end
   local input=con.getInputStream()
   local reader=BufferedReader(InputStreamReader(input,"UTF-8"))
   local sb=StringBuilder()
   local line=reader.readLine()
   while line~=nil do sb.append(line); line=reader.readLine() end
   reader.close(); con.disconnect()
   return tostring(sb.toString())
  end)
  handler.post(Runnable{run=function()
   if callback then callback(ok and result or nil) end
  end})
 end}).start()
end

function registrarPerfil(nombre,pais,callback)
 local clave=nombre:gsub("[%.%$%#%[%]/]","_")
 firebaseRequest("players/"..clave,"PUT",{
  username=nombre,country=pais,score=0,updatedAt=os.time()
 },function(r)
  if r then
   nombreUsuario=nombre; paisUsuario=pais
   editor.putString("user_name",nombre); editor.putString("user_country",pais); editor.commit()
   if callback then callback(true) end
  else if callback then callback(false) end end
 end)
end

function leerComentarios()
 firebaseRequest("comments","GET",nil,function(r)
  if not r then hablar("No se pudieron cargar los comentarios"); return end
  local ok,data=pcall(function() return cjson.decode(r) end)
  if not ok or type(data)~="table" then hablar("Todavía no hay comentarios"); return end
  local filas={}
  for _,v in pairs(data) do
   if type(v)=="table" then
    table.insert(filas,tostring(v.username or "Jugador").." ("..tostring(v.country or "País desconocido").."): "..tostring(v.text or ""))
   end
  end
  table.sort(filas)
  local msg=#filas>0 and table.concat(filas,"\n\n") or "Todavía no hay comentarios"
  local d=AlertDialog.Builder(ctx).setTitle("Comentarios de jugadores").setMessage(msg).setPositiveButton("Cerrar",nil).create()
  mostrarDialogoSeguro(d)
 end)
end

function leerEstadisticas()
 local opciones={"Ver jugadores top 100","Ver jugadores por país"}
 local principal=AlertDialog.Builder(ctx).setTitle("Estadísticas de jugadores")
 .setItems(opciones,DialogInterface.OnClickListener{
  onClick=function(d,which)
   if which==0 then
    firebaseRequest("players","GET",nil,function(r)
     if not r then hablar("No se pudieron cargar las estadísticas"); return end
     local ok,data=pcall(function() return cjson.decode(r) end)
     if not ok or type(data)~="table" then return end
     local arr={}
     for _,v in pairs(data) do
      if type(v)=="table" and tonumber(v.score or 0)>=100000 then table.insert(arr,v) end
     end
     table.sort(arr,function(a,b) return (tonumber(a.score) or 0)>(tonumber(b.score) or 0) end)
     local out={}
     for i=1,math.min(100,#arr) do table.insert(out,tostring(arr[i].username or "Jugador")..": "..tostring(arr[i].score or 0).." caramelos") end
     local resultado=AlertDialog.Builder(ctx).setTitle("Jugadores top 100").setMessage(#out>0 and table.concat(out,"\n") or "Aún no hay jugadores con 100 000 caramelos o más").setPositiveButton("Cerrar",nil).create()
     mostrarDialogoSeguro(resultado)
    end)
   else
    firebaseRequest("players","GET",nil,function(r)
     if not r then hablar("No se pudieron cargar los países"); return end
     local ok,data=pcall(function() return cjson.decode(r) end)
     if not ok or type(data)~="table" then return end
     local set,lista={},{}
     for _,v in pairs(data) do if type(v)=="table" and v.country then set[v.country]=true end end
     for k in pairs(set) do table.insert(lista,k) end
     table.sort(lista)
     if #lista==0 then hablar("Todavía no hay países registrados"); return end
     local paises=AlertDialog.Builder(ctx).setTitle("Jugadores por país").setItems(lista,DialogInterface.OnClickListener{
      onClick=function(dd,idx)
       local paisElegido=lista[idx+1]
       local out={}
       for _,v in pairs(data) do
        if type(v)=="table" and v.country==paisElegido then
         table.insert(out,tostring(v.username or "Jugador")..": "..tostring(v.score or 0).." caramelos")
        end
       end
       table.sort(out)
       local resultado=AlertDialog.Builder(ctx).setTitle(paisElegido).setMessage(#out>0 and table.concat(out,"\n") or "Sin jugadores").setPositiveButton("Cerrar",nil).create()
       mostrarDialogoSeguro(resultado)
      end
     }).create()
     mostrarDialogoSeguro(paises)
    end)
   end
  end
 }).create()
 mostrarDialogoSeguro(principal)
end
